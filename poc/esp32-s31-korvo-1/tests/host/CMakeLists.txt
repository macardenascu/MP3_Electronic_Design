cmake_minimum_required(VERSION 3.16)
project(music_catalog_host_tests C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_executable(music_catalog_test ../../main/music_catalog.c music_catalog_test.c)
target_include_directories(music_catalog_test PRIVATE include ../../main)
target_compile_definitions(music_catalog_test PRIVATE MUSIC_CATALOG_OPENDIR=music_catalog_test_opendir)

# Portable player core (main/core): pure logic, no peripherals.
set(CORE_DIR ../../main/core)
set(CORE_TESTS volume mute_sequencer power_policy player_core tone_gen wav_parser)
foreach(name IN LISTS CORE_TESTS)
    add_executable(${name}_test ${name}_test.c ${CORE_DIR}/${name}.c)
    target_include_directories(${name}_test PRIVATE include ${CORE_DIR})
    target_compile_options(${name}_test PRIVATE -Wall -Wextra -Werror)
endforeach()
target_sources(player_core_test PRIVATE ${CORE_DIR}/volume.c)
if(NOT MSVC)
    target_link_libraries(tone_gen_test PRIVATE m) # sin()/lround()
endif()

# Korvo board audio capabilities (main/korvo_audio_caps.c).
add_executable(korvo_audio_caps_test korvo_audio_caps_test.c ../../main/korvo_audio_caps.c)
target_include_directories(korvo_audio_caps_test PRIVATE include ../../main)
target_compile_options(korvo_audio_caps_test PRIVATE -Wall -Wextra -Werror)

# Korvo volume mapping and simulated battery (E5/E7).
add_executable(korvo_volume_test korvo_volume_test.c ../../main/korvo_volume.c)
target_include_directories(korvo_volume_test PRIVATE include ../../main)
target_compile_options(korvo_volume_test PRIVATE -Wall -Wextra -Werror)
add_executable(battery_sim_test battery_sim_test.c ${CORE_DIR}/battery_sim.c ${CORE_DIR}/power_policy.c)
target_include_directories(battery_sim_test PRIVATE include ${CORE_DIR})
target_compile_options(battery_sim_test PRIVATE -Wall -Wextra -Werror)

# MP3 decoder wrapper (main/decoder) around the vendored, unmodified minimp3.
set(DECODER_DIR ../../main/decoder)
set(MINIMP3_DIR ../../main/third_party/minimp3)
foreach(name IN ITEMS mp3_decoder_test mp3_to_wav)
    add_executable(${name} ${name}.c ${DECODER_DIR}/mp3_decoder.c)
    target_include_directories(${name} PRIVATE include ${DECODER_DIR} ${MINIMP3_DIR})
    target_compile_options(${name} PRIVATE -Wall -Wextra -Werror)
    if(NOT MSVC)
        target_link_libraries(${name} PRIVATE m)
    endif()
endforeach()
